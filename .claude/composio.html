issues = run_composio_tool(
  'SENTRY_LIST_ISSUES',
  project='api-prod',
  status='unresolved'
)

for issue in issues:
  issue['trace'] = run_composio_tool(
    'SENTRY_GET_EVENT',
    issue_id=issue['id']
  )

ranked = invoke_llm(
  f'Classify these {len(issues)} errors.'
  ' Return P0/P1/P2 with reasoning.'
)

for error in ranked['P0']:
  run_composio_tool(
    'LINEAR_CREATE_ISSUE',
    title=error['title'],
    team_id='ENG',
    priority=1
  )
